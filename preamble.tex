% Code blocks: wrap rather than run off the page.
%
% Pandoc defines Highlighting as a plain fancyvrb Verbatim, which does not
% break long lines, so anything wider than the text block silently ran into
% the margin and off the paper. fvextra adds breaklines. This file is
% included after pandoc's own definitions, so redefining here wins.
%
% At \small a code line fits about 73 characters. _common.R sets
% options(width = 72) so console output wraps before it reaches that, and
% breaklines catches whatever is still too long (long strings, wide tables).
\usepackage{fvextra}
\fvset{
  breaklines=true,
  breakanywhere=true,
  breakindent=1.5em,
  fontsize=\small
}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{%
  commandchars=\\\{\},%
  breaklines=true,%
  breakanywhere=true,%
  breakindent=1.5em,%
  fontsize=\small%
}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{%
  breaklines=true,%
  breakanywhere=true,%
  breakindent=1.5em,%
  fontsize=\small%
}

% Inline \texttt{} (package and function names) may break at the margin
% instead of overrunning it.
\emergencystretch=3em

% Contents: the book class allots 2.3em to a section number, which is not
% enough for a two-digit chapter with a two-digit section. "11.10" ran into
% its title with no space. Widen the number boxes at every level.
\makeatletter
\renewcommand*\l@section{\@dottedtocline{1}{1.5em}{3.0em}}
\renewcommand*\l@subsection{\@dottedtocline{2}{4.5em}{3.8em}}
\renewcommand*\l@subsubsection{\@dottedtocline{3}{8.3em}{4.8em}}
\makeatother
