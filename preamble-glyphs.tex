% Characters Latin Modern does not have.
%
% LuaLaTeX drops a character the current font lacks without an error. Only a
% "Missing character" line in the log records it, and the PDF shows a gap. The
% July build lost 647 glyphs this way. They were check marks and crosses in
% tables, a warning sign, >= and <= typed as Unicode in prose, Greek in code
% comments, and the bar characters of the inline histograms that skimr and
% similar summaries print. Most of these come from R output, so they cannot
% be fixed in the .Rmd text.
%
% Each character below is sent to DejaVu (free, and shipped with TeX) or, for
% relations, to the math font. Code blocks get DejaVu Sans Mono, so a bar in
% a histogram keeps the width of the code around it.
%
% Main book only. The Springer volume has none of these characters, and its
% preamble.tex is shared with this book, so this file stays out of it.
\usepackage{newunicodechar}
\newfontfamily\glyphsans{DejaVuSans.ttf}[Scale=MatchLowercase]
\newfontfamily\glyphmono{DejaVuSansMono.ttf}[Scale=MatchLowercase]

\makeatletter
% #1 is the code point. Use the mono face when the text around it is mono.
\DeclareRobustCommand\glyphfallback[1]{%
  {\edef\@tempa{\ttdefault}%
   \ifx\f@family\@tempa\glyphmono\else\glyphsans\fi
   \symbol{#1}}}
% Symbols with no mono form worth having. Always the sans face.
\DeclareRobustCommand\glyphsymbol[1]{{\glyphsans\symbol{#1}}}
% Greek typed as Unicode. In math it is the math letter, as unicode-math would
% have made it. Anywhere else it comes from DejaVu.
\DeclareRobustCommand\glyphgreek[2]{\ifmmode#1\else\glyphfallback{#2}\fi}
\makeatother
